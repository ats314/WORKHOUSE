\begin{abstract}
We present a curvature--functional--analytic framework towards a
nonperturbative mass gap for continuum SU(3) Yang--Mills theory.
On each lattice scale we assume a gauge--projected convexity bound
for the Wilson action, stability of this convexity under a viscous
Hamilton--Jacobi (Riccati) flow, and stability of the associated
Log--Sobolev (LSI) constant under block--spin renormalization.
Under these analytic hypotheses, we obtain a uniform spectral gap
for the physical sector at all lattice scales.
Assuming in addition that the continuum limit satisfies the
Osterwalder--Schrader axioms, in particular reflection positivity
and OS--reconstructibility, we conclude the existence of a strictly
positive mass gap for the continuum Yang--Mills Hamiltonian.
\end{abstract}

\section{Gauge Geometry and Physical Projection}
% ============================================================

Let $\Lambda_4$ be a four--dimensional periodic lattice of side length $L$,
and let
\[
\mathcal M_L = SU(3)^{\Lambda_4}
\]
denote the configuration space of link variables $U=(U_\mu(x))$,
with $x\in\Lambda_4$ and directions $\mu\in\{0,1,2,3\}$.

\subsection{Gauge generator and physical subspace}

Let $\omega:\Lambda_4 \to \mathfrak{su}(3)$ be an infinitesimal gauge parameter.
The linearized gauge generator is
\[
(G_L(U)\omega)_\mu(x)
=
\omega(x)\,U_\mu(x)
-
U_\mu(x)\,\omega(x+\hat\mu) .
\]

We define the \emph{gauge subspace}
\[
\mathcal H_{\mathrm{gauge}}(U) = \mathrm{Im}(G_L(U))
\subset
T_U\mathcal M_L
\]
and the \emph{physical subspace}
\[
\mathcal H_{\mathrm{phys}}(U) = \ker(G_L(U)^\dagger)
\subset
T_U\mathcal M_L
\]
with respect to the natural $L^2$ inner product.

\subsection{Physical projector}

Define
\[
P_L(U)
 :=
 I - G_L(U)\,
 \big(G_L(U)^\dagger G_L(U)\big)^{-1}
 G_L(U)^\dagger ,
\]
where the inverse is taken on the orthogonal complement of the kernel
of $G_L(U)^\dagger G_L(U)$ (global gauge modes).
Then
\[
T_U\mathcal M_L
=
\mathcal H_{\mathrm{phys}}(U)
\;\oplus\;
\mathcal H_{\mathrm{gauge}}(U),
\qquad
P_L(U)^2 = P_L(U) = P_L(U)^\dagger .
\]

% ============================================================
\section{Gauge--Projected Convexity of the Wilson Action}
% ============================================================

The Wilson action on $\mathcal M_L$ is
\[
S_L(U)
=
\beta\sum_{p} \left(1 - \frac{1}{3}\mathrm{Re}\,\mathrm{Tr}\,U_p\right),
\]
where the sum is over plaquettes $p$.

\begin{lemma}[Gauge--Projected Coercivity]
\label{lemma:coercive}
There exist $C_W>0$ and a neighborhood $\mathcal N$ of the trivial field
such that for all $L$ and all $U\in\mathcal N$,
\begin{equation}
P_L(U)\,\nabla^2 S_L(U)\,P_L(U)
\;\succeq\;
C_W\,P_L(U).
\tag{2.1}
\end{equation}
\end{lemma}

\begin{proof}[Proof Sketch]
Gauge invariance implies $\nabla^2 S_L(U)\,G_L(U)\equiv 0$.
Thus only the restriction to $\mathcal H_{\mathrm{phys}}(U)$ is relevant.
The statement is an analytic coercivity hypothesis on that restriction.
\end{proof}

% ============================================================
\section{Local Log--Sobolev Inequality in the Physical Sector}
% ============================================================

Let
\[
\mu_L(dU) = Z_L^{-1}\exp(-S_L(U))\,dU
\]
be the Wilson Gibbs measure (restricted to $\mathcal N$).
Define the physical carré-du-champ
\[
\Gamma_U(f) = \|P_L(U)\nabla f(U)\|^2.
\]

\begin{theorem}[Gauge--Projected Local LSI]
\label{thm:lsi}
Under Lemma~\ref{lemma:coercive},
for all smooth $f$ supported in $\mathcal N$,
\begin{equation}
\mathrm{Ent}_{\mu_L}(f^2)
\;\le\;
\frac{2}{C_W}
\int \Gamma_U(f)\,d\mu_L(U).
\tag{3.1}
\end{equation}
In particular, the physical spectral gap satisfies $\lambda_1\ge C_W$.
\end{theorem}

\begin{proof}[Proof Sketch]
Consider the generator
$L f = \Delta f - \langle \nabla S_L,\nabla f\rangle$
restricted to $\mathcal H_{\mathrm{phys}}(U)$ via $P_L(U)$.
The Bakry--Émery curvature operator $\Gamma_2$ in the physical sector
satisfies $\Gamma_2\ge C_W\Gamma$ due to Lemma~\ref{lemma:coercive}.
Standard Bakry--Émery theory yields the LSI.
\end{proof}

% ============================================================
\section{Riccati Stability of Convexity}
% ============================================================

Consider the viscous Hamilton--Jacobi flow
\[
\partial_t \phi_t
=
\frac12\Delta\phi_t
-
\frac12\langle\nabla S_L,\nabla\phi_t\rangle
+
\frac12\|\nabla\phi_t\|^2,
\]
and define $S_{L,t} = S_L + \phi_t$.

\begin{corollary}[Riccati Preservation of Physical Convexity]
\label{cor:riccati}
Under Lemma~\ref{lemma:coercive}, there exists $t_0>0$ such that
for all $t\in[0,t_0]$ and all $U\in\mathcal N$,
\[
P_L(U)\,\nabla^2 S_{L,t}(U)\,P_L(U)
\;\succeq\;
C_W\,P_L(U),
\]
uniformly in $L$.
Consequently, the physical LSI with constant $C_W$ holds for the
evolved measure $\mu_{L,t}\propto e^{-S_{L,t}}$.
\end{corollary}

\begin{proof}[Proof Sketch]
The Hessian evolution equation is of Riccati type, with nonnegative
group curvature.
A scalar comparison for the minimal eigenvalue yields a lower bound
$C_W$ for small $t$.
Bakry--Émery then gives LSI for $\mu_{L,t}$.
\end{proof}

% ============================================================
\section{Stability Under Renormalization}
% ============================================================

Let $\mathcal R_L:\mathcal M_L\to\mathcal M_{L'}$ be a block-spin (coarse--graining) map
with Lipschitz constant $L_{\mathcal R}\le 1$.
Define $\mu_{L'} = \mathcal R_{L\#}\mu_L$.

\begin{theorem}[RG Stability of the Physical LSI]
\label{thm:rg}
Suppose $\mu_L$ satisfies the physical LSI with constant $C_W$.
Then $\mu_{L'}$ also satisfies a physical LSI with constant
$\rho_{L'}\ge C_W$.
\end{theorem}

\begin{proof}[Proof Sketch]
Apply the fine‐scale LSI to $f\circ\mathcal R_L$, use the Lipschitz
bound on $D\mathcal R_L$, then pass to the pushforward measure.
The LSI constant is preserved when $L_{\mathcal R}\le 1$.
\end{proof}

\section{Continuum Reconstruction Assumptions}
% ============================================================

To make contact with the continuum Yang--Mills theory and its
Hamiltonian $H_{\mathrm{YM}}$, we must assume a standard
Osterwalder--Schrader (OS) reconstruction framework.

We assume:

\begin{itemize}
\item[(OS1)] \textbf{Existence of Continuum Limits.}
There exists a scaling sequence $a_L\to 0$ and renormalization steps
$n_L$ such that the family of Euclidean lattice measures
$\mu_L^{(n_L)}$ converges in distribution to a continuum Euclidean
field measure $\mu_{\mathrm{cont}}$ on gauge fields over $\mathbb R^4$.

\item[(OS2)] \textbf{OS Axioms.}
The measure $\mu_{\mathrm{cont}}$ satisfies the Osterwalder--Schrader
axioms: Euclidean invariance, reflection positivity, symmetry, and
cluster properties.

\item[(OS3)] \textbf{Reflection Positivity.}
For any time-reflection $\theta$ and suitable functionals $F$,
\[
\int F \cdot \theta(F)\, d\mu_{\mathrm{cont}} \;\ge\; 0.
\]

\item[(OS4)] \textbf{OS Reconstruction.}
From $\mu_{\mathrm{cont}}$ one can reconstruct a Hilbert space
$\mathcal H_{\mathrm{YM}}$, a vacuum vector $\Omega$, and a
selfadjoint Hamiltonian $H_{\mathrm{YM}}$ such that
Euclidean correlation functions are the analytic continuation of
Minkowski vacuum expectation values.

\end{itemize}

These assumptions are standard in constructive QFT and separate the
analytic convexity/functional analysis on the lattice from the
continuum reconstruction step.

\section{Continuum Limit Mass Gap}
% ============================================================

Let $\mu_L^{(n)}$ denote the $n$--fold RG pushforward of $\mu_L$.

\begin{theorem}[Continuum Mass Gap under Curvature–Stable RG]
\label{thm:massgap}
Assume:
\begin{itemize}
\item[(H1)] gauge–projected coercivity (Lemma~\ref{lemma:coercive})
holds with constant $C_W>0$ uniformly in $L$;
\item[(H2)] Riccati stability (Corollary~\ref{cor:riccati}) holds
with the same $C_W$;
\item[(H3)] RG stability (Theorem~\ref{thm:rg}) holds at each scale;
\item[(OS1)--(OS4)] the continuum reconstruction assumptions above.
\end{itemize}
Then for every $L$ and every RG level $n$,
$\mu_L^{(n)}$ satisfies a physical LSI with constant at least $C_W$
and hence has spectral gap $\lambda_1 \ge C_W$ in the physical sector.

In the continuum limit $a_L\to 0$ along a suitable RG trajectory,
the reconstructed Yang--Mills Hamiltonian $H_{\mathrm{YM}}$ has
spectrum
\[
\mathrm{Spec}(H_{\mathrm{YM}})
\subset \{0\}\cup[m_{\mathrm{gap}},\infty),
\qquad
m_{\mathrm{gap}} \ge \sqrt{C_W}.
\]
\end{theorem}

\begin{proof}[Proof Sketch]
By Theorem~\ref{thm:lsi} and Corollary~\ref{cor:riccati},
the physical LSI with constant $C_W$ holds at the fine lattice scale
for small fields in $\mathcal N$.
By Theorem~\ref{thm:rg}, this constant is preserved under RG
coarse–graining.
Thus every $\mu_L^{(n)}$ has a uniform physical spectral gap
$\lambda_1\ge C_W$.

Assumptions (OS1)--(OS4) ensure the existence of a continuum limit
measure $\mu_{\mathrm{cont}}$ and a reconstructed Hamiltonian
$H_{\mathrm{YM}}$.
Uniform spectral gap bounds in the Euclidean setting transfer to
a positive mass gap in the Minkowski Hamiltonian spectrum.
\end{proof}
