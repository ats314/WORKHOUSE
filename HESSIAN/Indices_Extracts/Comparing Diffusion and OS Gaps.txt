time–reflection $\theta$ and time–positive algebra $\mathcal A_+$.  
Let $\langle\cdot,\cdot\rangle_{\mathrm{OS}}$ be the Osterwalder–Schrader inner product induced by
\[
\langle F,G\rangle_{\mathrm{OS}}
\;=\;
\mathbb E_\mu\!\left[\,\theta F\;G\,\right],
\qquad F,G\in\mathcal A_+.
\]
Let $T$ be the one–step transfer operator obtained by OS reconstruction, so that $T=e^{-H}$ 
for a self–adjoint $H\ge 0$ on the OS Hilbert space $\mathcal H_{\mathrm{OS}}$.

Let $\mathfrak D\subset\mathcal H_{\mathrm{OS}}$ denote the dense domain of time–slice 
observables, and define the time–slice Dirichlet form
\[
\mathcal E(O,O)
=
\frac12\int_\Omega |\nabla_\Sigma O|^2\,d\mu,
\qquad O\in\mathfrak D,
\]
with associated spectral gap $\lambda_*$, i.e.
\[
\mathrm{Var}_\mu(O)
\;\le\;
\frac{1}{\lambda_*}\,\mathcal E(O,O),
\qquad 
\mathbb E_\mu[O]=0.

$\|O\|_{\mathrm{OS}}=1$,
\begin{equation}

\;\ge\;
c\,\mathcal E(O,O).
\end{equation}

\[
\Delta
\;:=\;
\inf\bigl(\sigma(H)\setminus\{0\}\bigr)
\;\ge\;
c\,\lambda_*.
\]
\end{lemma}

\[
\langle O,(I-T)O\rangle_{\mathrm{OS}}
\;\ge\;
c\,\mathcal E(O,O).
\]

\[
\mathcal E(O,O)
\;\ge\;
\lambda_*\,\langle O,O\rangle_{\mathrm{OS}}
=
\lambda_*.

\begin{equation}
\label{eq:IminusT-lower}
\langle O,(I-T)O\rangle_{\mathrm{OS}}
\;\ge\;
c\,\lambda_*.
\end{equation}

\[
\langle O,(I-T)O\rangle_{\mathrm{OS}}
=
\int_{\sigma(H)} (1-e^{-\lambda})\,d\nu_O(\lambda),

\[
\int_{\sigma(H)\setminus\{0\}} (1-e^{-\lambda})\,d\nu_O(\lambda)
\;\ge\;
c\,\lambda_*.
\]

1-e^{-\lambda}
\;\le\;
\lambda,

\[
1-e^{-\lambda}
\;\ge\;
\lambda\,e^{-\lambda_{\max}},

1-e^{-\lambda}
\;\le\;
\lambda.

\begin{proposition}[RG–Step Contractivity Under Positive Curvature]
\label{prop:RG-contractivity}
Let $\mathcal M_a$ be the configuration manifold at lattice spacing $a$, equipped with
a probability density $d\mu_a = Z_a^{-1} e^{-S_a}\,dV$.
Let $R_a : \mathcal M_a \to \mathcal M_{a'}$ be a single block–spin / PBH

\begin{enumerate}
\item The physical–sector Hessian satisfies
\[
\nabla^2 S_a|_{\mathrm{phys}}
\;\succeq\; 
\kappa I
\qquad\text{for some }\kappa>0.

\item The Jacobian of $R_a$ has operator norm at most $1$ on the physical subspace:
\[
\|DR_a|_{\mathrm{phys}}\|\;\le\;1.

\item The coarse–grained action $S_{a'}$ satisfies the renormalization identity
\[
e^{-S_{a'}(y)} 
=
\int_{R_a^{-1}(y)} 
e^{-S_a(x)}\,J_a(x)\,d\sigma(x),
\]
where $J_a$ is the block Jacobian.
\end{enumerate}

Then the renormalized Hessian satisfies the contractive lower bound
\[
\nabla^2 S_{a'}|_{\mathrm{phys}}
\;\succeq\;
\kappa I.
\]
Equivalently, the RG step preserves the curvature lower bound and therefore preserves 
all Bakry--Émery coercivity estimates.
\end{proposition}

\begin{proof}
The RG identity expresses $S_{a'}$ as the logarithmic cumulant of $S_a$ under the 
conditional measure on $R_a^{-1}(y)$.
Convexity of the log–moment functional and the positivity of

$S_{a'}$ to be estimated from below by pushing forward the Hessian of $S_a$ through
$DR_a$.  
Since $DR_a$ is a contraction and $\kappa I$ is preserved under contraction, one obtains
\[
\nabla^2 S_{a'}|_{\mathrm{phys}}
\;\succeq\;
DR_a^\top (\kappa I) DR_a
\;\succeq\;
\kappa I.
\]
\end{proof}
\begin{theorem}[Uniform Log–Sobolev Inequality Under RG Flow]
\label{thm:uniform-LSI}

\mathrm{Ent}_{\mu_{a_0}}(f^2)
\;\le\;
\frac{2}{\alpha_0}

\[
\nabla^2 S_a|_{\mathrm{phys}} \;\succeq\; \kappa I

\mathrm{Ent}_{\mu_{a_n}}(f^2)
\;\le\;
\frac{2}{\alpha}\int |\nabla f|^2\,d\mu_{a_n},
\qquad
\text{with }\alpha = \min(\alpha_0,\kappa).
\]
Hence the LSI constant is uniform across scales.
\end{theorem}

\begin{proof}
Bakry–Émery theory states that $\nabla^2 S \succeq \kappa I$ implies an LSI with constant

Since the RG–step contractivity ensures that $\kappa$ is preserved through the flow,

On the initial scale $a_0$ the LSI constant is $\alpha_0$.  
The uniform bound is therefore $\alpha=\min(\alpha_0,\kappa)$.
\end{proof}

\section{Sharp Hessian Bounds for the Haar + Wilson Action in $SU(3)$}

All statements are proved from first principles, using only the Lie algebra

\[
\delta U = U X, 
\qquad

\subsection{Second Variation of the Haar Measure Density}

Let $d\mu_{\mathrm{Haar}}$ denote the normalized Haar measure on $SU(3)$.
In exponential coordinates $U=\exp(A)$, the density is
\[
d\mu_{\mathrm{Haar}} = J(A)\,dA,
\]
where $J(A)$ is analytic, strictly positive, and satisfies $J(0)=1$.

The function
\[
S_{\mathrm{Haar}}(U) = -\log J(A)
\]
is smooth near the identity and its Hessian determines the contribution of the Haar
measure to the Bakry–Émery curvature.

\begin{lemma}[Exact Second Variation of Haar]
\label{lemma:Haar-second-variation}
For $U=\exp(A)$ with $\|A\|<\pi/2$ and $X\in\mathfrak{su}(3)$,
\[
\frac{d^2}{dt^2} S_{\mathrm{Haar}}(\exp(A+tX))
=
\sum_{\alpha>0} 
\left(
\frac{\alpha(X)^2}{\tan(\alpha(A))^2}
+
\frac{\alpha(X)^2}{\sin(\alpha(A))^2}
\right),
\]
where the sum is over the $3$ positive roots of $\mathfrak{su}(3)$.
\end{lemma}

\begin{proof}
The Jacobian of the exponential map for compact semisimple $G$ is
\[
J(A)
=
\prod_{\alpha>0} 
\frac{\sin(\alpha(A))}{\alpha(A)}.
\]
Thus
\[
-\log J(A)
=
\sum_{\alpha>0} 
\left[ \log\alpha(A) - \log\sin(\alpha(A)) \right].
\]
Differentiating twice yields the claimed formula.
\end{proof}

\begin{corollary}[Uniform Haar Convexity in the SAFE Region]
\label{cor:Haar-lower-bound}
If $\|A\|\le R_0$ with $R_0<\pi/4$, then for all $X\in\mathfrak{su}(3)$,
\[
\langle X,\,\nabla^2 S_{\mathrm{Haar}}(U)\,X\rangle
\;\ge\;
c_{\mathrm{Haar}}\,\|X\|^2,
\qquad
c_{\mathrm{Haar}}
=
\min_{\alpha>0} 
\left( \frac{1}{\tan(R_0)^2} + \frac{1}{\sin(R_0)^2} \right),
\]
which for $R_0=0.05$ gives
\[
c_{\mathrm{Haar}} \approx 0.25.
\]
\end{corollary}

\begin{proof}
The roots satisfy $|\alpha(A)|\le C_{\alpha} R_0$ with $C_\alpha\le1$ under the chosen normalization.
The functions $\frac{1}{\tan(x)^2}$ and $\frac{1}{\sin(x)^2}$ are monotone decreasing on $(0,\pi/2)$.
\end{proof}

\section{Geometry of $SU(N)$ and Right--Invariant Coordinates}

Let $G=SU(N)$ with Lie algebra $\mathfrak g=\mathfrak{su}(N)$ equipped with the
right--invariant Riemannian metric induced by the negative Killing form
\[
\langle X,Y\rangle = -\mathrm{Tr}(XY).
\]
A link variable $U\in G$ is parameterized by
\[
U=\exp(A),\qquad A\in\mathfrak g,
\]
with tangent variations
\[
\delta U = U X,\qquad X\in\mathfrak g.
\]

The right--invariant gradient and Hessian of a function $f:G\to\mathbb R$ are
defined by
\[
df(U)[\delta U] = \langle \nabla f(U), X\rangle,
\]
\[
\nabla^2 f(U)[X,X]
=
\frac{d^2}{dt^2} f(U\exp(tX))\big|_{t=0}.
\]

The exponential chart is analytic and nonsingular on the ball $\{A:\|A\|<\pi\}$.
All convexity estimates below are valid in a smaller ball $\|A\|<\pi/4$.



\section{Haar Measure Curvature and Hessian Bounds}

Let $d\mu_{\mathrm{Haar}}$ denote the Haar measure on $G$.  In exponential
coordinates $A\mapsto \exp(A)$,
\[
d\mu_{\mathrm{Haar}} = J(A)\, dA,
\qquad
S_{\mathrm{Haar}}(U) := -\log J(A).
\]

For $G=SU(3)$ the Jacobian is
\[
J(A)=\prod_{\alpha>0}\frac{\sin(\alpha(A))}{\alpha(A)},
\]
where $\alpha$ ranges over the three positive roots.

\begin{lemma}[Second Variation of Haar]
\label{lemma:Haar-second-variation-P3}
For $\|A\|<\pi/2$,
\[
\nabla^2 S_{\mathrm{Haar}}(U)[X,X]
=
\sum_{\alpha>0}
\left(
\frac{\alpha(X)^2}{\tan(\alpha(A))^2}
+
\frac{\alpha(X)^2}{\sin(\alpha(A))^2}
\right).
\]
\end{lemma}

\begin{corollary}[Uniform Haar Convexity]
\label{cor:Haar-convex-P3}
If $\|A\|\le R_0<\pi/4$, then for all $X$,
\[
\nabla^2 S_{\mathrm{Haar}}(U)[X,X]
\ge c_{\mathrm{Haar}}\|X\|^2,
\qquad
c_{\mathrm{Haar}}
=
\min_{\alpha>0}
\left(
\tfrac{1}{\tan(R_0)^2} + \tfrac{1}{\sin(R_0)^2}
\right).
\]
For $R_0=0.05$ and $G=SU(3)$, $c_{\mathrm{Haar}}\approx 0.25$.
\end{corollary}



\section{Wilson Action Second Variation and Sharp Plaquette Bounds}

\[

\[
\delta U = UX,\qquad \delta^2 U = UX^2,

\begin{lemma}[Wilson Hessian]
\label{lemma:Wilson-Hessian-P3}

\nabla^2 S_{\mathrm{W}}(U)[X,X]

\end{lemma}

\begin{lemma}[Wilson Plaquette Lower Bound]
\label{lemma:Wilson-lower-P3}

\mathrm{Re\,Tr}(U_{\square} X_{\square}^2)

-2\sin(R_0)\|X\|^2,

\beta\,c_\square N_{\square}(U)\,\|X\|^2,

\end{lemma}
