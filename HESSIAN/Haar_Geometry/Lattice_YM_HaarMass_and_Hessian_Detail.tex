\title{Technical Note: Haar Measure, Exponential Coordinates,\\
and the Hessian Mass Term on $SU(N)$}
\author{supplement to Pillar L}
\date{\today}

\maketitle

\begin{abstract}
We give a self--contained derivation of the Jacobian of the exponential
map on $SU(N)$ and extract the quadratic term in the Haar measure
action.  This yields an explicit expression for the Haar mass
coefficient $c_0$ in the small--field expansion used in Pillar L.
\end{abstract}

\section{Lie--theoretic preliminaries}

Let $G=SU(N)$ with Lie algebra $\mathfrak{g}=\mathfrak{su}(N)$.  We fix
a basis $\{T^a\}_{a=1}^{N^2-1}$ satisfying
\[
  [T^a,T^b] = f^{abc} T^c,\qquad
  \mathrm{Tr}(T^a T^b) = \frac{1}{2}\delta^{ab},
\]
with real, totally antisymmetric structure constants $f^{abc}$.

The adjoint representation is given by
\[
  (\mathrm{ad}_A)^{ab}
  = f^{acb} A^c \quad\text{for}\quad
  A = A^c T^c.
\]
The Killing form is
\[
  \kappa(A,B)
  = \mathrm{Tr}_{\mathfrak{g}}(\mathrm{ad}_A\mathrm{ad}_B)
  = 2N\,\mathrm{Tr}(AB),
\]
and the associated inner product
$\langle A,B\rangle = -\mathrm{Tr}(AB)$ is proportional to $-\kappa$.

\section{Jacobian of the exponential map}

Consider the exponential map
\[
  \exp:\mathfrak{g}\to G,\qquad A\mapsto U=\exp(iagA),
\]
where $a>0$ is a small parameter (lattice spacing) and $g$ is the bare
coupling.  The differential $d\exp_A:\mathfrak{g}\to T_U G$ can be
expressed in terms of the adjoint operator
$\mathrm{ad}_{iagA}$ via the identity
\[
  d\exp_A
  = \exp(iagA)\circ
    \frac{1 - e^{-\mathrm{ad}_{iagA}}}{\mathrm{ad}_{iagA}}.
\]
The Jacobian determinant of $\exp$ at $A$ with respect to the
bi--invariant Riemannian volume on $G$ and the Lebesgue measure on
$\mathfrak{g}$ is
\[
  J(A)
  = \det_{\mathfrak{g}}\!\left(
      \frac{\sinh(\tfrac{\mathrm{ad}_{iagA}}{2})}
           {\tfrac{\mathrm{ad}_{iagA}}{2}}
    \right).
\]
This formula follows from the Weyl integration formula and can be
derived by diagonalising $\mathrm{ad}_A$ in a root decomposition; we
omit the standard details.

\begin{definition}
The \emph{Haar measure action} is
\[
  S_{\mathrm{Haar}}(A) = -\log J(A).
\]
\end{definition}

\section{Small--field expansion of $S_{\mathrm{Haar}}$}

Set $Y = \frac{\mathrm{ad}_{iagA}}{2}$.  The Taylor expansion of
$\sinh$ gives
\[
  \frac{\sinh Y}{Y} = \mathbb{I} + \frac{Y^2}{3!} + \frac{Y^4}{5!}
                      + \cdots.
\]
For small $A$ (hence small $Y$),
\[
  \log\frac{\sinh Y}{Y}
  = \frac{Y^2}{3!} - \frac{Y^4}{3\cdot 5!} + O(\|Y\|^6).
\]
Taking the trace over $\mathfrak{g}$,
\[
  S_{\mathrm{Haar}}(A)
  = -\mathrm{Tr}_{\mathfrak{g}}\log\frac{\sinh Y}{Y}
  = -\frac{1}{3!}\mathrm{Tr}_{\mathfrak{g}}(Y^2)
    + O(\|Y\|^4),
\]
where we used that odd powers of $Y$ have vanishing trace and that the
$Y^4$ term is of order $\|Y\|^4$.

Since $Y = \tfrac{1}{2}\mathrm{ad}_{iagA}$, we have
\[
  Y^2 = -\frac{a^2 g^2}{4}\,\mathrm{ad}_A^2,
\]
and thus
\[
  -\frac{1}{3!}\mathrm{Tr}_{\mathfrak{g}}(Y^2)
  = \frac{a^2 g^2}{24}\,\mathrm{Tr}_{\mathfrak{g}}(\mathrm{ad}_A^2).
\]

\begin{lemma}[Trace of $\mathrm{ad}_A^2$ on $\mathfrak{su}(N)$]
\label{lem:adA2-trace}
For $A\in\mathfrak{su}(N)$ with $A = A^a T^a$ one has
\[
  \mathrm{Tr}_{\mathfrak{g}}(\mathrm{ad}_A^2)
  = C_2(\mathrm{ad})\,\sum_a (A^a)^2,
\]
with $C_2(\mathrm{ad}) = N$ in the chosen normalisation, so that
\[
  \mathrm{Tr}_{\mathfrak{g}}(\mathrm{ad}_A^2)
  = 2N\,\mathrm{Tr}(A^2).
\]
\end{lemma}

\begin{proof}
By definition,
\[
  (\mathrm{ad}_A)^{ab} = f^{acb} A^c.
\]
Then
\[
  \mathrm{Tr}_{\mathfrak{g}}(\mathrm{ad}_A^2)
  = (\mathrm{ad}_A)^{ab}(\mathrm{ad}_A)^{ba}
  = f^{acb}f^{b d a} A^c A^d.
\]
The contraction $f^{acb}f^{b d a}$ is proportional to $\delta^{cd}$,
with proportionality constant $C_2(\mathrm{ad})$ by $SU(N)$--invariance.
Evaluating at a convenient $A$ (for example, along a Cartan generator)
shows that $C_2(\mathrm{ad}) = N$ in the present normalisation.
Moreover, by the trace normalisation,
\[
  \sum_a (A^a)^2 = 2\,\mathrm{Tr}(A^2),
\]
so the second identity follows.
\end{proof}

Combining the previous expressions we arrive at:

\begin{theorem}[Haar mass coefficient]
\label{thm:haar-mass-coeff-technical}
Let $A\in\mathfrak{su}(N)$ with $A=\sum_a A^a T^a$ and
$\|A\|^2 = \sum_a (A^a)^2$.  Then
\[
  S_{\mathrm{Haar}}(A)
  = c_0 a^2 g^2 \|A\|^2 + O(a^4\|A\|^4),\qquad
  c_0 = \frac{C_2(\mathrm{ad})}{24}.
\]
In particular, $c_0>0$ depends only on $N$ and is independent of the
lattice volume.
\end{theorem}

\begin{remark}
In the convention where $\|A\|^2 = -\mathrm{Tr}(A^2)$, the numerical
value of $c_0$ becomes $c_0 = \frac{N^2-1}{2N}$.  The choice of
normalisation of $A$ and of the generators $T^a$ alters $c_0$ by a fixed
multiplicative factor, but the sign and dependence on $N$ are robust.
\end{remark}

\section{Hessian at the identity}

Differentiating the expansion in Theorem~\ref{thm:haar-mass-coeff-technical}
twice at $A=0$ shows that the Hessian of $S_{\mathrm{Haar}}$ at the
identity is a positive multiple of the identity on $\mathfrak{g}$:
\[
  \mathrm{Hess}\,S_{\mathrm{Haar}}(0)
  = 2c_0 a^2 g^2\,\mathrm{Id}_{\mathfrak{g}}.
\]
On the lattice, with link variables $U_b=\exp(iagA_b)$, the total
Hessian at the configuration $U_b=I$ for all $b$ is
\[
  \mathrm{Hess}\,S_{\mathrm{Haar}}(0)
  = 2c_0 a^2 g^2\,\mathbb{I}_{\mathfrak{g}^{N_b}},
\]
where $\mathbb{I}_{\mathfrak{g}^{N_b}}$ is the identity operator on the
product of Lie algebras over bonds.

This justifies the terminology \emph{Haar mass term} for the quadratic
contribution of $S_{\mathrm{Haar}}$ in exponential coordinates and
provides the starting point for the global Hessian analysis in Pillar L.
\qedhere
